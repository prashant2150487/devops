# This workflow automates testing, building, and deploying your React app
# It triggers on pushes and pull requests to main/develop branches
name: React CI/CD Pipeline

on:
  push:
    branches: [ main, develop ]  # Runs when code is pushed to these branches
  pull_request:
    branches: [ main, develop ]  # Runs when PRs are opened against these branches

jobs:
  # ============================================================================
  # JOB 1: BUILD AND TEST
  # ============================================================================
  # This job ensures code quality before deployment by:
  # - Testing on multiple Node versions (compatibility check)
  # - Running linters to catch code style issues
  # - Executing unit tests to verify functionality
  # - Building the app to catch build-time errors
  # WHY: Catches bugs early, prevents broken code from reaching production
  # ============================================================================
  build-and-test:
    runs-on: ubuntu-latest  # Uses GitHub's Ubuntu virtual machine (free tier available)
    
    # Matrix strategy runs this job multiple times with different Node versions
    # WHY: Ensures your app works across different Node environments
    strategy:
      matrix:
        node-version: [18.x, 20.x]  # Tests on Node 18 and 20 (current LTS versions)
    
    steps:
    # STEP 1: Get the code from your repository
    # WHY: Actions run on fresh VMs, so we need to clone the repo first
    - name: Checkout code
      uses: actions/checkout@v4  # Official GitHub action to clone your repo
    
    # STEP 2: Install Node.js
    # WHY: React requires Node.js to run npm commands and build tools
    - name: Setup Node.js ${{ matrix.node-version }}
      uses: actions/setup-node@v4
      with:
        node-version: ${{ matrix.node-version }}  # Uses version from matrix
        cache: 'npm'  # Caches node_modules to speed up future runs
    
    # STEP 3: Install project dependencies
    # WHY: Need all packages from package.json to build and test
    - name: Install dependencies
      run: npm ci  # 'npm ci' is faster than 'npm install' and ensures clean install
      # WHY npm ci over npm install:
      # - Deletes node_modules before installing (clean slate)
      # - Uses exact versions from package-lock.json (reproducible builds)
      # - 2-3x faster in CI environments
    
    # STEP 4: Check code quality with ESLint
    # WHY: Catches syntax errors, bad practices, and style violations
    - name: Run linter
      run: npm run lint --if-present  # Only runs if 'lint' script exists
      # --if-present prevents failure if script is missing
    
    # STEP 5: Run automated tests
    # WHY: Verifies that your components and functions work as expected
    - name: Run tests
      run: npm test -- --coverage --watchAll=false
      # --coverage: Generates test coverage report (shows % of code tested)
      # --watchAll=false: Runs once and exits (default watch mode waits for changes)
      # WHY: In CI we want tests to run once and report results, not watch for changes
    
    # STEP 6: Build the production app
    # WHY: Verifies the app can be built successfully and creates optimized files
    - name: Build project
      run: npm run build
      # Creates optimized production build in 'build/' folder
      # Minifies JS/CSS, optimizes images, removes debug code
    
    # STEP 7: Save build files for later use
    # WHY: Allows deploy job to use these files without rebuilding
    - name: Upload build artifacts
      uses: actions/upload-artifact@v4
      with:
        name: build-${{ matrix.node-version }}  # Unique name per Node version
        path: build/  # Folder containing built files
        retention-days: 7  # Keeps artifacts for 7 days (saves storage)
      # WHY: Artifacts let you download builds or pass them to other jobs

  # ============================================================================
  # JOB 2: DEPLOY TO GITHUB PAGES
  # ============================================================================
  # This job publishes your app to GitHub Pages (free hosting) when:
  # - All tests pass (needs: build-and-test)
  # - Code is pushed to main branch (not PRs or other branches)
  # WHY: Automates deployment so you don't manually upload files
  # RESULT: Your app is live at https://username.github.io/repo-name
  # ============================================================================
  deploy:
    needs: build-and-test  # Waits for build-and-test to complete successfully
    # WHY: Only deploy if tests pass (prevents deploying broken code)
    
    runs-on: ubuntu-latest
    
    # Conditional: Only deploy on main branch pushes (not PRs or other branches)
    # WHY: You want staging/feature branches tested but not deployed
    if: github.ref == 'refs/heads/main' && github.event_name == 'push'
    
    # Permissions needed for GitHub Pages deployment
    # WHY: Security best practice - grant only necessary permissions
    permissions:
      contents: read      # Read repository files
      pages: write        # Deploy to GitHub Pages
      id-token: write     # Required for secure deployment verification
    
    steps:
    # STEP 1: Get the code again
    # WHY: Each job runs on a fresh VM, so we need to clone repo again
    - name: Checkout code
      uses: actions/checkout@v4
    
    # STEP 2: Install Node.js (only version 20 for deployment)
    # WHY: We only need one version to deploy (already tested on 18 & 20)
    - name: Setup Node.js
      uses: actions/setup-node@v4
      with:
        node-version: '20.x'  # Latest LTS version
        cache: 'npm'
    
    # STEP 3: Install dependencies again
    # WHY: Fresh VM doesn't have node_modules from previous job
    - name: Install dependencies
      run: npm ci
    
    # STEP 4: Build the app for production
    # WHY: Creates optimized files for deployment
    - name: Build project
      run: npm run build
      env:
        CI: true  # Tells Create React App to build in CI mode (warnings = errors)
        PUBLIC_URL: /${{ github.event.repository.name }}
        # WHY PUBLIC_URL: GitHub Pages serves from /repo-name/, not root
        # Without this, CSS/JS paths would break (looking for /static/ instead of /repo-name/static/)
    
    # STEP 5: Configure GitHub Pages
    # WHY: Prepares Pages settings for deployment
    - name: Setup Pages
      uses: actions/configure-pages@v4
    
    # STEP 6: Upload build files to Pages artifact storage
    # WHY: Pages deployment system needs files in a specific artifact format
    - name: Upload artifact
      uses: actions/upload-pages-artifact@v3
      with:
        path: ./build  # The folder with your built React app
    
    # STEP 7: Deploy to GitHub Pages
    # WHY: This actually publishes your app to the live URL
    - name: Deploy to GitHub Pages
      id: deployment
      uses: actions/deploy-pages@v4
      # After this completes, your app is live at:
      # https://your-username.github.io/your-repo-name/

# ============================================================================
# WORKFLOW SUMMARY
# ============================================================================
# 1. TRIGGER: Push or PR to main/develop
# 2. BUILD-AND-TEST: Runs on Node 18 & 20
#    - Installs dependencies
#    - Lints code (checks quality)
#    - Runs tests (verifies functionality)
#    - Builds app (creates production files)
# 3. DEPLOY: Only on main branch pushes after tests pass
#    - Rebuilds app with correct paths for GitHub Pages
#    - Deploys to https://username.github.io/repo-name
# 4. RESULT: Automated quality checks + automatic deployment
# ============================================================================
